import csv, json, decimal, sys
from decimal import Decimal, InvalidOperation

def is_valid_amount(val):
    try:
        d = Decimal(val)
    except Exception:
        return False
    if not d.is_finite():
        return False
    if d < 0:
        return False
    # Check for fractional cents: amount must be integer cents (no fractional part)
    # Since amounts are INR decimal strings, we assume they are in rupees with up to 2 decimal places.
    # Convert to cents: multiply by 100 and check if integer.
    # Use quantize to avoid floating errors.
    if d != d.quantize(Decimal('0.01')):
        return False
    return True

def to_cents(val):
    # val is Decimal with exactly two decimal places
    return int((val * 100).to_integral_value())

# Load sales.csv
sales = {}
rejected_sales = 0
duplicate_sales = 0
with open('/workspace/sales.csv', newline='') as f:
    reader = csv.DictReader(f)
    for row in reader:
        order_id = row.get('order_id', '').strip()
        amount_str = row.get('amount', '').strip()
        if not order_id or not amount_str:
            rejected_sales += 1
            continue
        if not is_valid_amount(amount_str):
            rejected_sales += 1
            continue
        amount = Decimal(amount_str)
        cents = to_cents(amount)
        if order_id in sales:
            duplicate_sales += 1
            continue
        sales[order_id] = cents

# Load payments.csv
payments = {}
rejected_payments = 0
duplicate_payments = 0
unmatched_payments = []
with open('/workspace/payments.csv', newline='') as f:
    reader = csv.DictReader(f)
    for row in reader:
        payment_id = row.get('payment_id', '').strip()
        order_id = row.get('order_id', '').strip()
        amount_str = row.get('amount', '').strip()
        if not payment_id or not amount_str:
            rejected_payments += 1
            continue
        if not is_valid_amount(amount_str):
            rejected_payments += 1
            continue
        amount = Decimal(amount_str)
        cents = to_cents(amount)
        if payment_id in payments:
            duplicate_payments += 1
            continue
        payments[payment_id] = {'order_id': order_id, 'amount': cents}

# Match payments to sales
rows = []
for order_id, sales_cents in sorted(sales.items()):
    paid_cents = 0
    # Find all payments that match this order_id
    for pid, pdata in payments.items():
        if pdata['order_id'] == order_id:
            paid_cents += pdata['amount']
    balance_cents = sales_cents - paid_cents
    if paid_cents == 0:
        status = 'unpaid'
    elif 0 < paid_cents < sales_cents:
        status = 'partial'
    elif paid_cents == sales_cents:
        status = 'paid'
    else:
        status = 'overpaid'
    rows.append({
        'order_id': order_id,
        'sales_cents': sales_cents,
        'paid_cents': paid_cents,
        'balance_cents': balance_cents,
        'status': status
    })
    # Remove matched payments from unmatched list
    # We'll handle unmatched later

# Build unmatched_payments list
unmatched = []
for pid, pdata in payments.items():
    if pdata['order_id'] not in sales:
        unmatched.append({
            'payment_id': pid,
            'order_id': pdata['order_id'],
            'amount_cents': pdata['amount']
        })
# Sort by payment_id
unmatched.sort(key=lambda x: x['payment_id'])

output = {
    'currency': 'INR',
    'rows': rows,
    'rejected_sales': rejected_sales,
    'rejected_payments': rejected_payments,
    'duplicate_sales': duplicate_sales,
    'duplicate_payments': duplicate_payments,
    'unmatched_payments': unmatched
}

with open('/workspace/reconciled.json', 'w') as f:
    json.dump(output, f, indent=2)
