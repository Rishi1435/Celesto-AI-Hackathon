import csv, json, decimal, sys
from decimal import Decimal, InvalidOperation

def is_valid_amount(val):
    try:
        d = Decimal(val)
    except Exception:
        return False
    if not d.is_finite():
        return False
    if d < 0:
        return False
    # Check if amount is in rupees with up to 2 decimal places (paise)
    # Convert to string to avoid floating issues
    s = str(d).strip()
    if '.' in s:
        # Ensure at most 2 decimal places
        decimal_places = len(s.split('.')[1])
        if decimal_places > 2:
            return False
    # Also ensure no more than 2 digits after decimal
    # Also ensure no extra trailing zeros? Not needed.
    return True

def to_cents(val):
    # Convert Decimal to integer cents (paise)
    d = Decimal(val)
    # Multiply by 100 and quantize to integer
    return int((d * 100).to_integral_value())

# Load sales
sales_by_id = {}
rejected_sales = 0
duplicate_sales = 0
valid_sales = []
with open('/workspace/sales.csv', newline='') as f:
    reader = csv.DictReader(f)
    for row in reader:
        order_id = row.get('order_id', '').strip()
        amount_str = row.get('amount', '').strip()
        if not order_id or not amount_str:
            rejected_sales += 1
            continue
        if not is_valid_amount(amount_str):
            rejected_sales += 1
            continue
        # Valid sale
        cents = to_cents(amount_str)
        if order_id in sales_by_id:
            duplicate_sales += 1
            continue
        sales_by_id[order_id] = cents
        valid_sales.append({'order_id': order_id, 'sales_cents': cents})

# Load payments
payments_by_id = {}
rejected_payments = 0
duplicate_payments = 0
valid_payments = []
with open('/workspace/payments.csv', newline='') as f:
    reader = csv.DictReader(f)
    for row in reader:
        payment_id = row.get('payment_id', '').strip()
        amount_str = row.get('amount', '').strip()
        order_id = row.get('order_id', '').strip()
        if not payment_id or not amount_str:
            rejected_payments += 1
            continue
        if not is_valid_amount(amount_str):
            rejected_payments += 1
            continue
        cents = to_cents(amount_str)
        if payment_id in payments_by_id:
            duplicate_payments += 1
            continue
        payments_by_id[payment_id] = {'order_id': order_id, 'amount_cents': cents}
        valid_payments.append({'payment_id': payment_id, 'order_id': order_id, 'amount_cents': cents})

# Build unmatched payments (valid payments where order_id not in sales_by_id)
unmatched_payments = []
for p in valid_payments:
    if p['order_id'] not in sales_by_id:
        unmatched_payments.append({
            'payment_id': p['payment_id'],
            'order_id': p['order_id'],
            'amount_cents': p['amount_cents']
        })
# Sort by payment_id
unmatched_payments.sort(key=lambda x: x['payment_id'])

# Now reconcile each sale
rows = []
for order_id, sales_cents in sorted(sales_by_id.items()):
    paid_cents = 0
    # Sum payments for this order_id (only those that are valid and not duplicate)
    for p in valid_payments:
        if p['order_id'] == order_id:
            paid_cents += p['amount_cents']
    balance_cents = sales_cents - paid_cents
    if paid_cents == 0:
        status = 'unpaid'
    elif 0 < paid_cents < sales_cents:
        status = 'partial'
    elif paid_cents == sales_cents:
        status = 'paid'
    else:
        status = 'overpaid'
    rows.append({
        'order_id': order_id,
        'sales_cents': sales_cents,
        'paid_cents': paid_cents,
        'balance_cents': balance_cents,
        'status': status
    })

output = {
    'currency': 'INR',
    'rows': rows,
    'rejected_sales': rejected_sales,
    'rejected_payments': rejected_payments,
    'duplicate_sales': duplicate_sales,
    'duplicate_payments': duplicate_payments,
    'unmatched_payments': unmatched_payments
}

with open('/workspace/reconciled.json', 'w') as f:
    json.dump(output, f, indent=2)
